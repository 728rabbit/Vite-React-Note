## 🧩 一、多欄位表單管理

    import { useState } from "react";
    
    export default function App() {
      const [form, setForm] = useState({
        username: "",
        email: "",
        password: ""
      });
    
      function handleChange(e) {
        const { name, value } = e.target;
        setForm({ ...form, [name]: value });
      }
    
      function handleSubmit(e) {
        e.preventDefault();
        console.log("送出的表單資料：", form);
      }
    
      return (
        <form onSubmit={handleSubmit}>
          <input
            name="username"
            value={form.username}
            onChange={handleChange}
            placeholder="使用者名稱"
          />
          <input
            name="email"
            value={form.email}
            onChange={handleChange}
            placeholder="Email"
          />
          <input
            type="password"
            name="password"
            value={form.password}
            onChange={handleChange}
            placeholder="密碼"
          />
          <button type="submit">送出</button>
        </form>
      );
    }


-   使用單一 `handleChange` 管理多個欄位
    
-   用 `name` 屬性對應 state key

## 🧩 二、簡單表單驗證

    function handleSubmit(e) {
      e.preventDefault();
      const { username, email, password } = form;
      if (!username || !email || !password) {
        alert("請填寫完整欄位");
        return;
      }
      if (!email.includes("@")) {
        alert("Email 格式錯誤");
        return;
      }
      console.log("送出表單：", form);
    }


-   基本驗證：空白 / 格式
    
-   可以用狀態 (`useState`) 顯示錯誤訊息在畫面上

## 🧩 三、動態欄位（可新增 / 刪除）

    const [emails, setEmails] = useState([""]);
    
    function handleEmailChange(index, value) {
      const newEmails = [...emails];
      newEmails[index] = value;
      setEmails(newEmails);
    }
    
    function addEmailField() {
      setEmails([...emails, ""]);
    }
    
    function removeEmailField(index) {
      setEmails(emails.filter((_, i) => i !== index));
    }


-   適合處理多個相同欄位，例如多個聯絡人、標籤或商品屬性
    
-   使用陣列管理動態欄位

## 🧩 四、結合 API 模擬送出

    function handleSubmit(e) {
      e.preventDefault();
      fetch("https://jsonplaceholder.typicode.com/posts", {
        method: "POST",
        body: JSON.stringify(form),
        headers: { "Content-Type": "application/json" }
      })
      .then(res => res.json())
      .then(data => console.log("回傳資料：", data))
      .catch(err => console.error(err));
    }


-   POST 表單資料到 API
    
-   headers 設定 `Content-Type: application/json`
    
-   回傳結果可做成功提示或清空表單
