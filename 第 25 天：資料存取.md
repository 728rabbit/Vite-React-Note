## 🧩 一、用 localStorage 儲存 todos

### Step 1️⃣

在 `TodoContext.jsx` 中初始化時讀取資料：
    
    // context/TodoContext.jsx
    import { createContext, useReducer, useEffect } from "react";
    import { todoReducer } from "../reducers/todoReducer";
    
    export const TodoContext = createContext();
    
    export default function TodoProvider({ children }) {
      // 初始化從 localStorage 取資料
      const initialTodos = JSON.parse(localStorage.getItem("todos")) || [];
      const [todos, dispatch] = useReducer(todoReducer, initialTodos);
    
      // 每次 todos 改變就寫回 localStorage
      useEffect(() => {
        localStorage.setItem("todos", JSON.stringify(todos));
      }, [todos]);
    
      return (
        <TodoContext.Provider value={{ todos, dispatch }}>
          {children}
        </TodoContext.Provider>
      );
    }



✅ 功能解釋：

-   初始載入時讀取 localStorage
    
-   每次 todos 改變就同步寫入
    
-   使用 `useEffect` 自動偵測變化

## 🧩 三、進階：用假 API 模擬後端存取

> 這部分是可選的，但非常實用！

我們使用 JSONPlaceholder 當假 API  
（或用你自己的 Laravel / PHP API 也可以）

### Step 1️⃣ 安裝 axios

    npm install axios


### Step 2️⃣ 建立簡單 API 模組

    // api/todoApi.js
    import axios from "axios";
    
    const API_URL = "https://jsonplaceholder.typicode.com/todos";
    
    export async function getTodos() {
      const res = await axios.get(`${API_URL}?_limit=5`);
      return res.data;
    }

----------

### Step 3️⃣ 在 `TodoContext.jsx` 載入 API

    import { createContext, useReducer, useEffect } from "react";
    import { todoReducer } from "../reducers/todoReducer";
    import { getTodos } from "../api/todoApi";
    
    export const TodoContext = createContext();
    
    export default function TodoProvider({ children }) {
      const [todos, dispatch] = useReducer(todoReducer, []);
    
      useEffect(() => {
        async function fetchData() {
          const data = await getTodos();
          dispatch({ type: "set", payload: data.map(t => ({
            id: t.id,
            text: t.title,
            done: t.completed
          })) });
        }
        fetchData();
      }, []);
    
      return (
        <TodoContext.Provider value={{ todos, dispatch }}>
          {children}
        </TodoContext.Provider>
      );
    }
    
----------

### Step 4️⃣ 在 Reducer 新增 “set” 行為

    // reducers/todoReducer.jsx
    export function todoReducer(state, action) {
      switch (action.type) {
        case "set":
          return action.payload;
        case "add":
          return [...state, { id: Date.now(), text: action.payload, done: false }];
        case "toggle":
          return state.map(todo =>
            todo.id === action.payload ? { ...todo, done: !todo.done } : todo
          );
        case "delete":
          return state.filter(todo => todo.id !== action.payload);
        default:
          return state;
      }
    }

## 🧩 四、選擇性延伸：同步儲存

你可以再用 axios 把：

-   新增任務（`POST`）
    
-   刪除任務（`DELETE`）
    
-   切換完成狀態（`PATCH`）
    

都對應到後端 API。  
但 **重點不是寫完所有 CRUD**，而是練習 **理解資料流與同步邏輯** 💪
