## 🧩 一、表單基本結構回顧

    import { useState } from "react";
    
    function App() {
      const [name, setName] = useState("");
    
      function handleSubmit(e) {
        e.preventDefault(); // 阻止頁面刷新
        if (!name.trim()) return alert("名字不能空白");
        alert(`你輸入的名字是：${name}`);
        setName("");
      }
    
      return (
        <form onSubmit={handleSubmit}>
          <input
            type="text"
            placeholder="輸入名字"
            value={name}
            onChange={(e) => setName(e.target.value)}
          />
          <button type="submit">送出</button>
        </form>
      );
    }
    
    export default App;

### 1️⃣ 多個輸入欄位表單

    import { useState } from "react";
    
    function App() {
      const [form, setForm] = useState({ username: "", password: "" });
    
      function handleChange(e) {
        const { name, value } = e.target;
        setForm({ ...form, [name]: value });
      }
    
      function handleSubmit(e) {
        e.preventDefault();
        if (!form.username || !form.password) return alert("請填寫完整欄位");
        alert(`登入成功：${form.username}`);
        setForm({ username: "", password: "" });
      }
    
      return (
        <form onSubmit={handleSubmit}>
          <input
            name="username"
            value={form.username}
            onChange={handleChange}
            placeholder="使用者名稱"
          />
          <input
            type="password"
            name="password"
            value={form.password}
            onChange={handleChange}
            placeholder="密碼"
          />
          <button type="submit">登入</button>
        </form>
      );
    }
    
    export default App

;

-   `name` 屬性幫助我們用一個 `handleChange` 管理多個欄位
    
-   `...form` 展開現有資料，再更新對應欄位

### 2️⃣ 鍵盤事件範例
 

    import { useState } from "react";
    
    function App() {
      const [text, setText] = useState("");
    
      function handleKeyDown(e) {
        if (e.key === "Enter") {
          alert(`你按下 Enter 並輸入了：${text}`);
        }
      }
    
      return (
        <input
          value={text}
          onChange={(e) => setText(e.target.value)}
          onKeyDown={handleKeyDown}
          placeholder="按 Enter 試試"
        />
      );
    }
    
    export default App;


-   `onKeyDown` 監聽鍵盤按鍵
    
-   可以針對 Enter、Esc 或其他按鍵做不同操作

### 3️⃣ 結合 checkbox / radio / select

    import { useState } from "react";
    
    function App() {
      const [checked, setChecked] = useState(false);
      const [color, setColor] = useState("red");
    
      return (
        <div>
          <label>
            <input
              type="checkbox"
              checked={checked}
              onChange={() => setChecked(!checked)}
            />
            同意條款
          </label>
    
          <select value={color} onChange={(e) => setColor(e.target.value)}>
            <option value="red">紅色</option>
            <option value="green">綠色</option>
            <option value="blue">藍色</option>
          </select>
    
          <p>checkbox: {checked ? "已勾選" : "未勾選"}</p>
          <p>選擇顏色: {color}</p>
        </div>
      );
    }
    
    export default App;


-   `checkbox` → 布林值
    
-   `select` → 取得選中值
    
-   可以搭配 state 直接控制畫面呈現
