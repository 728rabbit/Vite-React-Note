
## 🧠 一、什麼是列表渲染？

-   在 React 中，我們通常有一個陣列資料
    
-   用 `map()` 將每個資料渲染成 **組件或元素**
    
-   這樣可以動態生成多個畫面區塊
    

----------

## 🧩 二、練習 1：簡單陣列列表

    function App() {
      const fruits = ['蘋果', '香蕉', '橘子'];
    
      return (
        <div>
          <h1>水果清單</h1>
          <ul>
            {fruits.map((fruit, index) => (
              <li key={index}>{fruit}</li>
            ))}
          </ul>
        </div>
      );
    }
    
    export default App;

-   說明：
    
    -   `fruits.map(...)` → 遍歷陣列
        
    -   `key={index}` → React 追蹤每個元素的唯一識別
        
    -   結果會生成三個 `<li>` 元素

## 🧩 三、練習 2：列表加減

    import { useState } from  'react'

    function Fruitslist() {
	    
	    const [fruits, setFruits] = useState(['蘋果', '香蕉', '橘子']);
	    const [fruitName, setFruitName] = useState('');
	    
	    function addFruit() {
		    if((fruitName.toString().trim()) != '') {
			    setFruits([...fruits, fruitName]);
			    setFruitName('');
		    }
		    else {
			    alert('請先輸入水果名稱');
		    }
	    }
	    
	    function deleteFruit(indexToDelete = -1) {
		    const newFruits = fruits.filter((_, index) => index !== indexToDelete);
		    setFruits(newFruits);
	    }
	    
	    return (
		    <div>
			    <h1>水果清單</h1>
			    
			    <ul>
			    {fruits.map((fruit, index) => (
			    <li  key={index}><span>{fruit}</span> | <button  type="button"  onClick={() => deleteFruit(index)}>刪除</button></li>
			    ))}
			    </ul>
			    
			    <input  type="text"  placeholder="輸入水果名稱..."  value={fruitName}  onChange={(e) => setFruitName(e.target.value)}  />
			    
			    <button  type="button"  onClick={addFruit}>+添加水果</button>
		    </div>
	    );
    
    }
    export  default Fruitslist;
